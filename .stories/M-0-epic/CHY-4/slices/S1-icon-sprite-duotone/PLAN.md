```js
import icons from '@iconify-json/solar/icons.json' with { type: 'json' };
export const ICONS = [['i-regular-house', 'home-2-line-duotone'], /* …58 */];
export function buildSprite() {
  const syms = ICONS.toSorted(([a], [b]) => a.localeCompare(b)).map(([id, name]) => {
    const body = icons.icons[name]?.body ?? (() => { throw new Error(`no Solar icon for ${id}`); })();
    return `<symbol id="${id}" viewBox="0 0 24 24">${body.replace(/opacity="\.5"/g, 'style="opacity:var(--duo,.5);color:var(--duo-c,currentColor)"')}</symbol>`;
  });
  return `<!-- generated by pnpm icons — do not edit -->\n<svg …>${syms.join('')}</svg>\n`;
}
```
